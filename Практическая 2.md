```java
import java.util.Scanner;

public class Task01 {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Введите пароль: ");
        String password = scanner.nextLine();

        boolean hasUppercase = false;
        boolean hasDigit = false;
        boolean hasSpecial = false;

        for (char c : password.toCharArray()) {
            if (Character.isUpperCase(c)) {
                hasUppercase = true;
            }
            if (Character.isDigit(c)) {
                hasDigit = true;
            }
            if ("!@#$%^&*".indexOf(c) >= 0) {
                hasSpecial = true;
            }
        }

        boolean valid = password.length() >= 8
                && hasUppercase
                && hasDigit
                && hasSpecial;

        System.out.println(valid ? "Пароль надежный" : "Пароль не соответствует требованиям");
    }
}
```

### 6. Блокировщик брутфорса

```java
public class Task06 {
    static class LoginThrottler {
        private int failedAttempts = 0;
        private long blockedUntil = 0;

        public boolean canTry() {
            return System.currentTimeMillis() >= blockedUntil;
        }

        public void registerFailedAttempt() {
            failedAttempts++;

            if (failedAttempts >= 5) {
                blockedUntil = System.currentTimeMillis() + 60_000;
                failedAttempts = 0;
            }
        }

        public void reset() {
            failedAttempts = 0;
        }

        public long getRemainingBlockSeconds() {
            long remaining = blockedUntil - System.currentTimeMillis();
            return Math.max(0, remaining / 1000);
        }
    }

    public static void main(String[] args) {
        LoginThrottler throttler = new LoginThrottler();

        for (int i = 1; i <= 5; i++) {
            if (throttler.canTry()) {
                System.out.println("Попытка " + i);
                throttler.registerFailedAttempt();
            }
        }

        System.out.println("Осталось блокировки: "
                + throttler.getRemainingBlockSeconds() + " сек.");
    }
}
```

```java
    public static String encrypt(String text, int shift) {
        StringBuilder result = new StringBuilder();

        for (char c : text.toCharArray()) {
            result.append((char) (c + shift));
        }
```

```java
import java.util.*;

public class Task12 {
    static class PaginationHelper {
        public static <T> List<T> getPage(List<T> items, int page, int pageSize) {
            if (page < 1 || pageSize <= 0) {
                return Collections.emptyList();
            }

            int start = (page - 1) * pageSize;

            if (start >= items.size()) {
                return Collections.emptyList();
            }

            int end = Math.min(start + pageSize, items.size());

            return new ArrayList<>(items.subList(start, end));
        }
    }

    public static void main(String[] args) {
        List<String> news = new ArrayList<>();

        for (int i = 1; i <= 55; i++) {
            news.add("Новость " + i);
        }

        List<String> page = PaginationHelper.getPage(news, 2, 20);

        for (String item : page) {
            System.out.println(item);
        }
    }
}
```

### 17. Удаление свайпом с возможностью отмены (Undo)

```java
import java.util.*;

public class Task17 {
    static class UndoBuffer<T> {
        private T deletedItem;
        private long deleteTime;
        private final long undoTimeMs = 5000;

        public void delete(T item) {
            deletedItem = item;
            deleteTime = System.currentTimeMillis();
        }

        public T undo() {
            if (deletedItem != null
                    && System.currentTimeMillis() - deleteTime <= undoTimeMs) {
                T item = deletedItem;
                deletedItem = null;
                return item;
            }

            deletedItem = null;
            return null;
        }
    }

    public static void main(String[] args) {
        UndoBuffer<String> buffer = new UndoBuffer<>();

        buffer.delete("Удаленный элемент");

        String restored = buffer.undo();

        System.out.println(restored != null
                ? "Восстановлено: " + restored
                : "Время отмены истекло");
    }
}
```

```java
import java.util.*;

public class Task20 {
    static class ImageFile {
        String name;
        long size;

        ImageFile(String name, long size) {
            this.name = name;
            this.size = size;
        }
    }

    public static void main(String[] args) {
        Queue<ImageFile> cache = new LinkedList<>();
        long totalSize = 0;
        long maxSize = 100L * 1024 * 1024;

        ImageFile[] images = {
                new ImageFile("image1.jpg", 40L * 1024 * 1024),
                new ImageFile("image2.jpg", 40L * 1024 * 1024),
                new ImageFile("image3.jpg", 40L * 1024 * 1024)
        };

        for (ImageFile image : images) {
            cache.add(image);
            totalSize += image.size;

            while (totalSize > maxSize && !cache.isEmpty()) {
                ImageFile oldest = cache.poll();
                totalSize -= oldest.size;
                System.out.println("Удален старый файл: " + oldest.name);
            }
        }

        System.out.println("Размер кэша: " + totalSize / (1024 * 1024) + " МБ");
    }
}
```

### 41. Расчет расстояния между двумя GPS-точками

```java
public class Task41 {
    public static double haversine(
            double lat1, double lon1,
            double lat2, double lon2) {

        final double EARTH_RADIUS = 6_371_000;

        double lat1Rad = Math.toRadians(lat1);
        double lat2Rad = Math.toRadians(lat2);
        double deltaLat = Math.toRadians(lat2 - lat1);
        double deltaLon = Math.toRadians(lon2 - lon1);

        double a = Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2)
                + Math.cos(lat1Rad) * Math.cos(lat2Rad)
                * Math.sin(deltaLon / 2) * Math.sin(deltaLon / 2);

        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

        return EARTH_RADIUS * c;
    }

    public static void main(String[] args) {
        double distance = haversine(
                55.7558, 37.6173,
                59.9343, 30.3351
        );

        System.out.printf("Расстояние: %.2f м%n", distance);
    }
}
```

```java
public class Task42 {
    public static double distance(
            double lat1, double lon1,
            double lat2, double lon2) {

        final double R = 6_371_000;

        double dLat = Math.toRadians(lat2 - lat1);
        double dLon = Math.toRadians(lon2 - lon1);

        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2)
                + Math.cos(Math.toRadians(lat1))
                * Math.cos(Math.toRadians(lat2))
                * Math.sin(dLon / 2) * Math.sin(dLon / 2);
```

### 49. Плеер аудиофайлов (State Machine)

```java
public class Task49 {
    enum State {
        IDLE, INITIALIZED, PREPARED, PLAYING, PAUSED, STOPPED
    }

    static class AudioPlayer {
        private State state = State.IDLE;

        public void initialize() {
            if (state == State.IDLE) {
                state = State.INITIALIZED;
                printState();
            }
        }

        public void prepare() {
            if (state == State.INITIALIZED) {
                state = State.PREPARED;
                printState();
            }
        }

        public void play() {
            if (state == State.PREPARED || state == State.PAUSED) {
                state = State.PLAYING;
                printState();
            } else {
                System.out.println("Недопустимый переход: PLAY");
            }
        }

        public void pause() {
            if (state == State.PLAYING) {
                state = State.PAUSED;
                printState();
            } else {
                System.out.println("Недопустимый переход: PAUSE");
            }
        }

        public void stop() {
            if (state == State.PLAYING || state == State.PAUSED) {
                state = State.STOPPED;
                printState();
            } else {
                System.out.println("Недопустимый переход: STOP");
            }
        }

        private void printState() {
            System.out.println("Состояние: " + state);
        }
    }

    public static void main(String[] args) {
        AudioPlayer player = new AudioPlayer();

        player.initialize();
        player.prepare();
        player.play();
        player.pause();
        player.play();
        player.stop();
    }
}
```

### №4. Повторные попытки с задержкой (Retry Pattern)

```dart
Future<T> retry<T>(Future<T> Function() task, int maxAttempts) async {
  Object? lastError;
  for (var attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      print('Попытка $attempt');
      return await task();
    } catch (e) {
      lastError = e;
      if (attempt < maxAttempts) {
        await Future.delayed(const Duration(seconds: 1));
      }
    }
  }
  throw lastError ?? Exception('Задача не выполнена');
}

void main() async {
  var calls = 0;
  try {
    final result = await retry(() async {
      calls++;
      if (calls < 3) throw Exception('Временная ошибка');
      return 'Успешный результат';
    }, 3);
    print(result);
  } catch (e) {
    print('Все попытки завершились ошибкой: $e');
  }
}
```

### №2. Экспоненциальная задержка (Exponential Backoff)

```dart
class SocketException implements Exception {
  final String message;
  const SocketException(this.message);
  @override
  String toString() => message;
}

Future<T> retryRequest<T>(Future<T> Function() request, int attempts) async {
  for (var i = 0; i < attempts; i++) {
    try { return await request(); }
    on SocketException {
      if (i == attempts - 1) rethrow;
      final seconds = 1 << i;
      print('Повтор через $seconds сек.');
      await Future.delayed(Duration(seconds: seconds));
    }
  }
  throw Exception('Запрос не выполнен');
}

void main() async {
  var count = 0;
  try {
    final result = await retryRequest(() async {
      count++;
      if (count < 3) throw const SocketException('Нет соединения');
      return 'Сервер ответил';
    }, 4);
    print(result);
  } catch (e) { print(e); }
}
```

```dart
import 'dart:async';

Stream<T> throttle<T>(Stream<T> source, Duration interval) {
  late StreamController<T> output;
  var allowed = true;
  StreamSubscription<T>? sub;
  output = StreamController<T>(onListen: () {
    sub = source.listen((value) {
      if (!allowed) return;
      allowed = false;
      output.add(value);
      Timer(interval, () => allowed = true);
    }, onError: output.addError, onDone: output.close);
  }, onCancel: () => sub?.cancel());
  return output.stream;
}

void main() async {
  final clicks = Stream.periodic(const Duration(milliseconds: 200), (i) => i + 1).take(10);
  await for (final click in throttle(clicks, const Duration(seconds: 1))) {
    print('Принят клик №$click');
  }
}
```

```dart
class DataCache {
  final Map<String, String> memory = {};
  final Map<String, String> disk = {'user': 'Данные с диска'};
  Future<String> get(String key) async {
    if (memory.containsKey(key)) { print('Источник: память'); return memory[key]!; }
    await Future.delayed(const Duration(milliseconds: 100));
    if (disk.containsKey(key)) { print('Источник: диск'); return memory[key] = disk[key]!; }
    await Future.delayed(const Duration(milliseconds: 500));
    print('Источник: сеть');
    return memory[key] = 'Данные из сети для $key';
  }
}
```

### №50. Реактивный модуль финансового терминала

```dart
import 'dart:async';
import 'dart:math';

class FinanceTerminal {
  final _prices = StreamController<double>();
  final _screen = StreamController<String>.broadcast();
  Timer? _timer;
  Stream<String> get screen => _screen.stream;
  void connect() {
    final random = Random(4);
    var price = 65000.0;
    _timer = Timer.periodic(const Duration(milliseconds: 100), (_) {
      price += (random.nextDouble() - 0.5) * 100;
      if (!_prices.isClosed) _prices.add(price);
    });
    final history = <double>[];
    var lastUpdate = DateTime.fromMillisecondsSinceEpoch(0);
    _prices.stream.listen((value) {
      history.add(value);
      if (history.length > 5) history.removeAt(0);
      final now = DateTime.now();
      if (now.difference(lastUpdate).inMilliseconds >= 17) {
        lastUpdate = now;
        final average = history.reduce((a, b) => a + b) / history.length;
        _screen.add('Цена: ${value.toStringAsFixed(2)} | среднее: ${average.toStringAsFixed(2)}');
      }
    });
  }
  Future<void> dispose() async {
    _timer?.cancel();
    await _prices.close();
    await _screen.close();
  }
}

void main() async {
  final terminal = FinanceTerminal();
  final sub = terminal.screen.listen(print);
  terminal.connect();
  await Future.delayed(const Duration(seconds: 2));
  await sub.cancel();
  await terminal.dispose();
  print('Терминал закрыт');
}
```
